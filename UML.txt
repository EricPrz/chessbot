Class: Game
  - board: Board
  - turn: Color
  - castling: CastlingRights
  - en_passant: Position
  - halfmove_clock: int
  - fullmove_number: int
  - parent: Game (optional)
  - move_made: Move (optional)
  
  + Game()                              // Starting position
  + Game(fen: String)                   // From FEN
  + get_legal_moves(): [Game]           // All next states
  + get_legal_moves_uci(): [String]     // For debugging
  + is_check(): bool
  + is_checkmate(): bool
  + is_stalemate(): bool
  + is_draw(): bool
  + get_fen(): str
  + get_pgn(): str
  + get_nnue_encoding(): HalfKPEncoding // For NN input
  + copy(): Game

Class: Board
  - grid: [8][8] of Piece (or null)
  - kings: {Color: Position}
  
  + get_piece(pos): Piece
  + set_piece(pos, Piece)
  + move_piece(from, to): Piece (returns captured)
  + is_occupied(pos): bool
  + is_attacked(pos, by_color): bool
  + find_king(color): Position
  + copy(): Board

Class: Piece
  - type: PieceType
  - color: Color
  + get_pseudo_legal_moves(pos, board): [Position]
  + copy(): Piece

Class: Move
  - from: Position
  - to: Position
  - piece: Piece
  - captured: Piece (optional)
  - promotion: PieceType (optional)
  - is_castle: bool
  - is_en_passant: bool
  
  + to_uci(): str
  + to_san(board): str

Class: CastlingRights
  - white_kingside: bool
  - white_queenside: bool
  - black_kingside: bool
  - black_queenside: bool
  
  + copy(): CastlingRights

Enum: PieceType
  - PAWN, KNIGHT, BISHOP, ROOK, QUEEN, KING
  + from_char(char): PieceType
  + to_char(): char
  + get_value(): int

Enum: Color
  - WHITE, BLACK
  + opposite(): Color
  + to_char(): char
